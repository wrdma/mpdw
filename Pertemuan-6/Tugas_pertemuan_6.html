Penanganan ketidakstasioneran

bc.fwd <- function(x, lam) if (abs(lam) < 0.05) log(x) else (x^lam - 1) / lam
bc.inv <- function(y, lam) if (abs(lam) < 0.05) exp(y) else (lam * y + 1)^(1 / lam)

if (!ragam_stasioner) {
  cat("Ragam belum stasioner -> transformasi Box-Cox, lambda =", lambda_opt, "\n")
  train.t <- ts(bc.fwd(train.ts, lambda_opt))
  plot.ts(train.t, main = "Data Latih Setelah Transformasi")
} else {
  cat("Ragam sudah stasioner -> tanpa transformasi\n")
  lambda_opt <- 1
  train.t <- train.ts
}

fit_model <- function(x, p, q) {
  tryCatch(Arima(x, order = c(p, 0, q), method = "ML"), error = function(e) NULL)
}

sig_semua <- function(m) {
  ct <- lmtest::coeftest(m)
  pv <- ct[!rownames(ct) %in% "intercept", 4]
  if (length(pv) == 0) return(NA)
  isTRUE(all(pv < 0.05))
}
